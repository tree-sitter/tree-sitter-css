  &:where([data-theme="midnight"] *) {

    &:hover {

        (block
          (at_rule
            (at_keyword))))))

==============================

  /* before */ &:where(.dark, .dark *) /* after */
);
@theme {
  --color-brand: red;
}
.button {
  color: blue;
}

---

  (import_statement

    (selectors

          (class_selector
            (class_name
              (identifier)))

            (class_selector

                (identifier)))
